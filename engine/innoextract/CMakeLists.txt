# innoextract (pinned, with myiosdeck.patch), the Boost sources it needs and MYIOSDECK's C API
# (myiosdeck_inno.cpp) as one static library. engine/innoextract/build.sh configures this for
# iOS arm64 (the app) or for the build machine (the CI test, with test/unpack_test.c).
cmake_minimum_required(VERSION 3.16)
project(myiosdeck_inno C CXX)

set(IX_SRC "" CACHE PATH "innoextract checkout")
set(BOOST_SRC "" CACHE PATH "Boost source tree (boost_1_83_0)")
set(LZMA_INCLUDE "" CACHE PATH "Directory with lzma.h (empty: the system's)")
set(INNO_SHA "unknown" CACHE STRING "innoextract commit")
option(MID_INNO_TEST "Also build test/unpack_test" OFF)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
endif()

# What innoextract's own CMake would detect on Linux, macOS and iOS alike. No process spawning:
# unrar is not run (GOG's old RAR installers are reported instead).
set(GEN ${CMAKE_BINARY_DIR}/gen)
file(WRITE ${GEN}/configure.hpp "// Generated by engine/innoextract/CMakeLists.txt
#ifndef INNOEXTRACT_CONFIGURE_HPP
#define INNOEXTRACT_CONFIGURE_HPP
#define INNOEXTRACT_HAVE_ISATTY 1
#define INNOEXTRACT_HAVE_IOCTL 1
#define INNOEXTRACT_HAVE_TIMEGM 1
#define INNOEXTRACT_HAVE_GMTIME_R 1
#define INNOEXTRACT_HAVE_UTIMENSAT 1
#define INNOEXTRACT_HAVE_DYNAMIC_UTIMENSAT 0
#define INNOEXTRACT_HAVE_AT_FDCWD 1
#define INNOEXTRACT_HAVE_UTIMES 1
#define INNOEXTRACT_HAVE_DLSYM 0
#define INNOEXTRACT_HAVE_BUILTIN_BSWAP16 1
#define INNOEXTRACT_HAVE_BUILTIN_BSWAP32 1
#define INNOEXTRACT_HAVE_BUILTIN_BSWAP64 1
#define INNOEXTRACT_HAVE_BSWAP_16 0
#define INNOEXTRACT_HAVE_BSWAP_32 0
#define INNOEXTRACT_HAVE_BSWAP_64 0
#define INNOEXTRACT_HAVE_ALIGNOF 1
#define INNOEXTRACT_HAVE_STD_CODECVT_UTF8_UTF16 0
#define INNOEXTRACT_HAVE_STD_UNIQUE_PTR 1
#define INNOEXTRACT_HAVE_DECRYPTION 1
#define INNOEXTRACT_HAVE_LZMA 1
#define INNOEXTRACT_HAVE_ICONV 1
#define INNOEXTRACT_HAVE_WIN32_CONV 0
#define INNOEXTRACT_HAVE_BUILTIN_CONV 0
#define INNOEXTRACT_HAVE_POSIX_SPAWNP 0
#define INNOEXTRACT_HAVE_UNISTD_ENVIRON 0
#define INNOEXTRACT_HAVE_FORK 0
#define INNOEXTRACT_HAVE_EXECVP 0
#define INNOEXTRACT_HAVE_WAITPID 0
#endif
")
file(WRITE ${GEN}/release.cpp "// Generated by engine/innoextract/CMakeLists.txt
#include \"release.hpp\"
const char innoextract_name[] = \"innoextract\";
const char innoextract_version[] = \"1.10-dev+${INNO_SHA} (MYIOSDECK)\";
const char innosetup_versions[] = \"Inno Setup 1.2.10 to 6.4.0\";
const char innoextract_bugs[] = \"https://innoextract.constexpr.org/issues\";
const char innoextract_copyright[] = \"Copyright (C) 2011-2020 Daniel Scharrer <daniel@constexpr.org>\";
const char innoextract_license[] = \"zlib license\";
")

file(GLOB IX_SOURCES
	${IX_SRC}/src/cli/*.cpp ${IX_SRC}/src/crypto/*.cpp ${IX_SRC}/src/loader/*.cpp
	${IX_SRC}/src/setup/*.cpp ${IX_SRC}/src/stream/*.cpp ${IX_SRC}/src/util/*.cpp)
list(REMOVE_ITEM IX_SOURCES
	${IX_SRC}/src/cli/main.cpp      # the command line (Boost.ProgramOptions)
	${IX_SRC}/src/cli/debug.cpp     # DEBUG builds only
	${IX_SRC}/src/util/test.cpp     # innoextract's unit test runner
	${IX_SRC}/src/util/windows.cpp) # Windows only
file(GLOB BOOST_FS_SOURCES ${BOOST_SRC}/libs/filesystem/src/*.cpp)
set(BOOST_IO_SOURCES
	${BOOST_SRC}/libs/iostreams/src/zlib.cpp
	${BOOST_SRC}/libs/iostreams/src/bzip2.cpp
	${BOOST_SRC}/libs/iostreams/src/file_descriptor.cpp)
if(NOT IX_SOURCES OR NOT BOOST_FS_SOURCES)
	message(FATAL_ERROR "IX_SRC (${IX_SRC}) and BOOST_SRC (${BOOST_SRC}) must point at the sources")
endif()

add_library(myiosdeck_inno STATIC
	${IX_SOURCES} ${GEN}/release.cpp
	${BOOST_FS_SOURCES} ${BOOST_IO_SOURCES}
	${CMAKE_CURRENT_SOURCE_DIR}/myiosdeck_inno.cpp)
target_include_directories(myiosdeck_inno PRIVATE
	${GEN} ${IX_SRC}/src ${CMAKE_CURRENT_SOURCE_DIR}/../../App/Sources/Native)
target_include_directories(myiosdeck_inno SYSTEM PRIVATE ${BOOST_SRC})
if(LZMA_INCLUDE)
	target_include_directories(myiosdeck_inno SYSTEM PRIVATE ${LZMA_INCLUDE})
endif()
target_compile_definitions(myiosdeck_inno PRIVATE
	NDEBUG MYIOSDECK_NO_PROCESSES BOOST_ALL_NO_LIB BOOST_SYSTEM_DISABLE_THREADS
	BOOST_FILESYSTEM_STATIC_LINK=1 BOOST_IOSTREAMS_STATIC_LINK=1
	MYIOSDECK_INNO_SHA="${INNO_SHA}")
target_compile_options(myiosdeck_inno PRIVATE -O2 -g0 -Wno-deprecated-declarations)
set_source_files_properties(${BOOST_FS_SOURCES} PROPERTIES
	COMPILE_DEFINITIONS BOOST_FILESYSTEM_SOURCE COMPILE_OPTIONS -w)
set_source_files_properties(${BOOST_IO_SOURCES} PROPERTIES
	COMPILE_DEFINITIONS BOOST_IOSTREAMS_SOURCE COMPILE_OPTIONS -w)

if(MID_INNO_TEST)
	add_executable(unpack_test ${CMAKE_CURRENT_SOURCE_DIR}/test/unpack_test.c)
	target_include_directories(unpack_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../App/Sources/Native)
	set_target_properties(unpack_test PROPERTIES LINKER_LANGUAGE CXX)
	target_link_libraries(unpack_test myiosdeck_inno lzma bz2 z)
	if(APPLE)
		target_link_libraries(unpack_test iconv)
	endif()
endif()
